async function loadLeagueData(leagueId) { try { // Calling relative URL '/api/league/474413' routes the request through server.js const response = await fetch(`/api/league/${leagueId}`); if (!response.ok) { throw new Error(`HTTP Error: ${response.status}`); } const data = await response.json(); console.log('League Data Loaded:', data); // Process standings and display on UI displayStandings(data.standings.results); } catch (error) { console.error('Failed to load league data:', error); } }
